Files used described below:

  • Bison_Genotypes_Raw(CORRECTED).csv

Remove sex-linked loci and allele errors

Sex-linked loci can be excluded prior to analyses as to not artificially increase profile completeness metrics in subsequent data quality screening. Sample “Wab24_011” has three alleles called for locus “BMS601”.

# Locus BM6017 and BMS911 are X-linked, and locus SE4748 is a sex marker
wb_locus_cleaned <- wb_raw_herds %>% 
  select(-BM6017, -BM6017_2, -BMS911, -BMS911_2, -SE4748, -SE4748_2) %>% 
  # Fix locus calling errors: 
  mutate(
    BMS601   = if_else(Sample.ID == "Wab24_011", NA, BMS601),
    BMS601_2 = if_else(Sample.ID == "Wab24_011", NA, BMS601_2)
  )

Sample quality and duplicate removal

Calculate genotype profile completeness

# Many samples were collected in duplicate (ie. tissue and fecal)
# Assign a profile completeness metric to the sample to keep the highest covered sample
 
usuable_data <- usuable_data %>%
  # Convert "", "**", "0", "-99" to NA for all character columns
  mutate(across(
    where(is.character),
    ~ replace(.x, .x %in% c("", "**", "0", "-99"), NA)
  )) %>%
  # Calculates a row sum of non-NA values (% of profile amplified)
  mutate(COMPLETE = rowSums(!is.na(across(14:ncol(usuable_data))))) %>%
  # Moves newly created column with profile completeness (%) after AMPLIFICATION
  relocate(COMPLETE, .after = AMPLIFICATION)
 
# The "COMPLETE" column represents the coverage across the (now) 50 autosomal loci

Filter low quality samples

Prepare samples for duplicate genotype detection

To use the dupGenotypes function in strataG, data must be in a gtypes object which can be converted from a genind object.

# Transform the dataframe into a genind then a gtypes object. First, combine locus columns
locus_cols <- names(wb_cleaned_ids)[14:ncol(wb_cleaned_ids)]
 
# Get base locus names without "_2"
base_loci <- unique(gsub("_2$", "", locus_cols))
 
# Loop through each locus and combine alleles
for (locus in base_loci) {
  col1 <- locus
  col2 <- paste0(locus, "_2")
  if (col1 %in% names(wb_cleaned_ids) && col2 %in% names(wb_cleaned_ids)) {
    wb_cleaned_ids[[col1]] <- paste(wb_cleaned_ids[[col1]], wb_cleaned_ids[[col2]], sep = "/")
  }
}
 
# Remove all the _2 columns
wb_cleaned_ids <- wb_cleaned_ids[, !grepl("_2$", names(wb_cleaned_ids))]
 
# Clean missing data values
wb_cleaned_ids[wb_cleaned_ids == "0/0"] <- NA
wb_cleaned_ids[wb_cleaned_ids == "**/**"] <- NA
wb_cleaned_ids[wb_cleaned_ids == "NA/NA"] <- NA
wb_cleaned_ids[wb_cleaned_ids == "-99/-99"] <- NA

# Assign a "Spreadsheet.ID" Column so each sample has unique name
wb_cleaned_ids <- wb_cleaned_ids %>% 
  mutate(Spreadsheet.ID = row_number()) %>% 
  relocate(Spreadsheet.ID, .before = Year.Genotyped)
 
# Import as a gtypes object with strataG package
wb.genind <- df2genind(X=wb_cleaned_ids[,c(15:ncol(wb_cleaned_ids))],
                              sep = "/",
                              ind.names = wb_cleaned_ids$Spreadsheet.ID,
                              loc.names = NULL,
                              pop = wb_cleaned_ids$Herd,
                              NA.char = "NA",
                              ploidy = 2,
                              type = "codom",
                              strata = NULL,
                              hierarchy = NULL)
 
wb.gtypes <- genind2gtypes(wb.genind)

Identify duplicate genotype profiles

# Make a df with the duplicates called with 90% or more profile similarity (ie. 45 or more exact loci)
# This can take a while on slow computers
dupe_profiles <- dupGenotypes(wb.gtypes, 
                  num.shared = 0.90)

Remove duplicate profiles from dataset

wb_non_dup_ids_screened <- wb_cleaned_ids %>% 
  mutate(Spreadsheet.ID = as.character(Spreadsheet.ID)) %>% 
  anti_join(
    dupe_profiles, 
    by = c("Spreadsheet.ID" = "ids.2", "Herd" = "strata.2")
  )

Generate final sample IDs
